## `src/audio/chip.js` (append-only)

```js
/** World 1 · Level 9 — Sheetwater: canon in three octaves (one voice per sheet). */
const SHEET_STEP = 60 / 72 / 4
export function scheduleSheetwater(step, t, buf) {
  const canon = (voice, phase) => {
    const u = step + phase
    if (u % 16 === 0) blipAt(buf, t, 220 * Math.pow(2, voice / 12), 0.08, 'sine', 0.35)
    if (u % 8 === 2) hatAt(buf, t, 0.04, 0.12)
    if (u % 16 === 4) padAt(buf, t, 330 * Math.pow(2, voice / 12), 0.5, 0.08)
    if (u % 16 === 12) blipAt(buf, t, 440 * Math.pow(2, voice / 12), 0.06, 'triangle', 0.28)
  }
  canon(0, 0)
  canon(12, 4)
  canon(-12, 8)
}
export function startSheetwaterMusic() {
  startLoop(scheduleSheetwater, SHEET_STEP)
}
export function sfxStackSwap() {
  blipAt(globalBuf(), audioCtx.currentTime, 520, 0.12, 'square', 0.2)
}
```

export function LiftlingSpawn({ e }) {
  const g = useRef()
  useFrame(({ clock }) => {
    g.current.position.y = 0.55 + Math.sin(clock.elapsedTime * 4 + e.seed) * 0.08
  })
  return (
    <group ref={g} position={[e.x, 0.55, e.z]}>
      <mesh>
        <octahedronGeometry args={[0.28, 0]} />
        <meshStandardMaterial {...obsidian()} emissive={glow()} emissiveIntensity={1.4} toneMapped={false} />
      </mesh>
    </group>
  )
}

export function StackSentinelSpawn({ e, maxHp }) {
  const g = useRef()
  useFrame(({ clock }) => {
    g.current.rotation.y = clock.elapsedTime * 0.5
  })
  return (
    <group ref={g} position={[e.x, 0, e.z]}>
      <mesh position={[0, 0.9, 0]}>
        <cylinderGeometry args={[0.35, 0.45, 1.6, 8]} />
        <meshStandardMaterial {...obsidian()} emissive={glow()} emissiveIntensity={0.9} />
      </mesh>
      <mesh position={[0, 1.7, 0]}>
        <coneGeometry args={[0.25, 0.5, 8]} />
        <meshStandardMaterial color="#ffb347" emissive="#ffb347" emissiveIntensity={1.2} toneMapped={false} />
      </mesh>
    </group>
  )
}
