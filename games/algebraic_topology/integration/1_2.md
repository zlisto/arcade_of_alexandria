## `src/audio/chip.js` (append-only)

Bastion march 104 bpm G major + bell on loop close (`sfxLoopClose`).

```javascript
// --- Bouquet Bastion theme (World 1) + loop bell ---
const BASTION_STEP = 60 / 104 / 4
const BASTION_CHORDS = [
  [43, [67, 71, 74, 79]],
  [40, [64, 67, 71, 76]],
  [38, [62, 66, 69, 74]],
  [43, [67, 71, 74, 79]],
]
function scheduleBastion(step, t, buf) {
  const s = step % 64
  const bar = Math.floor(s / 16)
  const k = s % 16
  const [root, tones] = BASTION_CHORDS[bar % 4]
  if (k % 2 === 0) blipAt(t, hz(root - 12), { dur: BASTION_STEP * 1.4, type: 'square', vol: 0.04, bus: musicBus })
  if (k === 0 || k === 8) padAt(t, tones, BASTION_STEP * 8, 0.06)
  if (k === 4) blipAt(t, hz(79), { dur: 0.08, type: 'triangle', vol: 0.05, bus: musicBus })
}
export function startBastionMusic() {
  startLoop(scheduleBastion, BASTION_STEP)
}
export function sfxLoopClose() {
  const t = audio()?.currentTime ?? 0
  blipAt(t, hz(76), { dur: 0.35, type: 'sine', vol: 0.14, bus: sfxBus })
  blipAt(t + 0.08, hz(67), { dur: 0.5, type: 'triangle', vol: 0.1, bus: sfxBus })
}
```

In `Level1_2.jsx`, swap `startForgeMusic` → `startBastionMusic` and call `sfxLoopClose()` where `sfxForgeEdge` marks a closed loop.
