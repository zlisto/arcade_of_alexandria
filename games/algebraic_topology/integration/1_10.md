## `src/audio/chip.js` (append)

```javascript
/** World 1 · Level 10 — rotating round (three voices, 96 bpm). */
const DECK_STEP = 60 / 96 / 4
export function scheduleDeckRound(step, t, buf) {
  const beat = Math.floor(t / DECK_STEP)
  if (beat % 12 === 0) blipAt(buf, t, 220, 0.08, 'sine', 0.35)
  if (beat % 6 === 3) hatAt(buf, t, 0.04)
  if (beat % 24 === 0) padAt(buf, t, [220, 277, 330], 0.5, 0.12)
  const voice = (beat >> 2) % 3
  if (beat % 4 === voice) blipAt(buf, t, 440 + voice * 55, 0.06, 'triangle', 0.22)
}
export function startDeckRoundMusic() {
  startLoop(scheduleDeckRound, DECK_STEP)
}
export function sfxDeckShift() {
  blip(330, 0.08, 'sine', 0.4)
}
```
