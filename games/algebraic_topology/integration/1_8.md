## `src/audio/chip.js` (append)

Sheetwater theme — 92 bpm, three staggered glass pads + harp arpeggios; layer mutes when lift sheet changes (hook in level optional).

```js
const SHEET_STEP = 60 / 92 / 4
function scheduleSheetwater(step, t, buf) {
  const bar = Math.floor(t / 16) % 4
  hatAt(buf, t, 0.04, 5200)
  if (t % 4 === 0) blipAt(buf, t, 55, 0.22, 'sine')
  ;[0, 5, 10].forEach((off, i) => {
    const root = [261.63, 329.63, 392][i]
    padAt(buf, t + off * step, root, 0.08, 0.35)
  })
  if (bar < 3) blipAt(buf, t + 8 * step, rootFromBar(bar), 0.12, 'triangle')
}
function rootFromBar(b) {
  return [523.25, 659.25, 783.99, 987.77][b]
}
export function startSheetwaterMusic() {
  startLoop(scheduleSheetwater, SHEET_STEP)
}
```

In `Level1_8.jsx`, replace `startTideMusic()` with `startSheetwaterMusic()`.
