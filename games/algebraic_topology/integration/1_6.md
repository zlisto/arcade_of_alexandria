## 5. `src/characters/Spawn.jsx` — `PatrolSpawn` (append export)

```jsx
export function PatrolSpawn({ e, maxHp = 4 }) {
  const g = useRef()
  useFrame(({ clock }) => {
    if (!e.alive) return
    g.current.position.set(e.x, 0.55 + Math.sin(clock.elapsedTime * 2 + e.seed) * 0.04, e.z)
    g.current.rotation.y = e.face
  })
  if (!e.alive && e.dying <= 0) return null
  return (
    <group ref={g}>
      <mesh scale={e.dying ? 1 - e.dying : 1}>
        <capsuleGeometry args={[0.35, 0.9, 6, 12]} />
        <meshStandardMaterial color="#1a1028" emissive="#8060d0" emissiveIntensity={1.1} toneMapped={false} />
      </mesh>
      <mesh position={[0.42, 0.35, 0]} rotation={[0, 0, 0.4]}>
        <torusGeometry args={[0.28, 0.06, 8, 20]} />
        <meshStandardMaterial color="#8060d0" emissive="#b090ff" emissiveIntensity={1.4} toneMapped={false} />
      </mesh>
    </group>
  )
}
```
