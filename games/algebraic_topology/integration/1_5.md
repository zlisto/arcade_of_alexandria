## `src/audio/chip.js` (append-only)

Rolling 6/8 march, two interlocking arpeggios (tube / hole):

```javascript
/** World 1 · Level 5 — Torus Terraces (108 bpm, 6/8 feel). */
const PUSH_STEP = 60 / 108 / 4
export function schedulePush(step, t, buf) {
  const beat = Math.floor(t / PUSH_STEP)
  if (beat % 12 === 0) blipAt(buf, t, 73.42, 0.24, 'sine', 0.09)
  if (beat % 6 === 0) hatAt(buf, t, 0.045)
  if (beat % 12 === 3) padAt(buf, t, 220, 0.14, 0.32)
  if (beat % 6 === 1) blipAt(buf, t, 329.63, 0.12, 'triangle', 0.05)
  if (beat % 6 === 4) blipAt(buf, t, 246.94, 0.11, 'triangle', 0.05)
}
export function startPushMusic() {
  startLoop(schedulePush, PUSH_STEP)
}

export function sfxProject() {
  blipAt(getBuf(), audioCtx.currentTime, 523.25, 0.22, 'sine', 0.11)
}
```

Optional: call `sfxProject()` in `Level1_5.jsx` when projector fires.

## `src/characters/Spawn.jsx` (append)

```jsx
/** World 1 · Level 5 — hijacks a map projector until lashed. */
export function ProjectorImpSpawn({ e, maxHp = 1 }) {
  const g = useRef()
  useFrame(({ clock }) => {
    if (!e.alive) return
    g.current.position.y = 0.75 + Math.sin(clock.elapsedTime * 4 + e.seed) * 0.08
    g.current.rotation.y = clock.elapsedTime * 1.2
  })
  if (!e.alive) return null
  return (
    <group ref={g} position={[e.x, 0.75, e.z]}>
      <mesh>
        <sphereGeometry args={[0.28, 10, 8]} />
        <meshStandardMaterial color="#1a1424" emissive="#8060d0" emissiveIntensity={1.2} toneMapped={false} />
      </mesh>
      <mesh position={[0, 0.22, 0]}>
        <coneGeometry args={[0.12, 0.28, 6]} />
        <meshStandardMaterial color="#6040a0" emissive="#b090ff" emissiveIntensity={1.4} toneMapped={false} />
      </mesh>
    </group>
  )
}
```

Add to spawn switch: `projector_imp: ProjectorImpSpawn`.
