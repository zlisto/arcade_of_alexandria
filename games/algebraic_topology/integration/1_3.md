## `src/audio/chip.js` (append-only)

```javascript
/** World 1 · Level 3 — rising ostinato; step up each lap, chime on floor (104 bpm E minor). */
const HELIX_STEP = 60 / 104 / 4
export function scheduleHelix(step, t, buf) {
  const beat = Math.floor(t / HELIX_STEP)
  if (beat % 16 === 0) blipAt(buf, t, 82, 0.22, 'sine', 0.08)
  if (beat % 4 === 0) hatAt(buf, t, 0.04)
  if (beat % 8 === 2) padAt(buf, t, 164.81, 0.12, 0.35)
  if (beat % 16 === 12) blipAt(buf, t, 329.63, 0.18, 'triangle', 0.06)
}
export function startHelixMusic() {
  startLoop(scheduleHelix, HELIX_STEP)
}

export function sfxFloor() {
  blipAt(getBuf(), audioCtx.currentTime, 880, 0.2, 'sine', 0.12)
}
export function sfxGate() {
  blipAt(getBuf(), audioCtx.currentTime, 440, 0.25, 'triangle', 0.1)
}
```

Wire `sfxFloor` on lap bank and `sfxGate` on gate open in `Level1_3.jsx` when merging.

## `src/characters/Spawn.jsx` (append model + export)

```jsx
export function WinderSpawn({ e }) {
  const g = useRef()
  useFrame(({ clock }) => {
    if (!e.alive) return
    g.current.rotation.y = clock.elapsedTime * 0.7
  })
  if (!e.alive) return null
  return (
    <group ref={g} position={[e.x, 0.9, e.z]}>
      <mesh>
        <cylinderGeometry args={[0.32, 0.38, 1.6, 10]} />
        <meshStandardMaterial color="#1c1a22" roughness={0.9} />
      </mesh>
      <mesh rotation={[Math.PI / 2, 0, 0]}>
        <torusGeometry args={[0.55, 0.07, 8, 28]} />
        <meshStandardMaterial color="#6040a0" emissive="#9060ff" emissiveIntensity={1.3} toneMapped={false} />
      </mesh>
    </group>
  )
}
```

In the spawn switch, add `winder: WinderSpawn`.
