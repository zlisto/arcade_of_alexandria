## `src/audio/chip.js` (append)

```js
const BASIN_STEP = 60 / 108 // ~108 bpm low drone + march triplet
function scheduleBasin(step, t, buf) {
  const beat = step % 16
  if (beat === 0) blipAt(buf, t, 55, 0.22, 'sine', 0.35)
  if (beat === 8) blipAt(buf, t, 82, 0.12, 'triangle', 0.2)
  if (beat % 4 === 2) hatAt(buf, t, 0.08)
  if (step % 32 === 0) padAt(buf, t, 130, 0.5, 0.06)
}
export function startBasinMusic() {
  startLoop(scheduleBasin, BASIN_STEP)
}
export function sfxSeize() {
  blipAt(getBuf(), audioCtx.currentTime, 40, 0.4, 'sawtooth', 0.25)
}
```
