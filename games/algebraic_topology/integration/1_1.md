## `src/audio/chip.js` — Bouquet Bastion theme (append)

Stately 104 bpm G major march (design `world-1.json` 1-1 music):

```js
const BASTION_STEP = 60 / 104 / 4
function scheduleBastion(step, t, buf) {
  const beat = Math.floor(t / step) % 16
  if (beat % 4 === 0) blipAt(buf, t, 98, 0.22, 'sine', 0.35)
  if (beat % 8 === 4) hatAt(buf, t, 0.06)
  if (beat % 2 === 0) padAt(buf, t, [55, 62, 67], 0.08, 0.4)
  const mel = [67, 69, 71, 74, 71, 69, 67, 62][Math.floor(beat / 2) % 8]
  if (beat % 2 === 1) blipAt(buf, t, mel, 0.12, 'square', 0.18)
}
export function startBastionMusic() {
  startLoop(scheduleBastion, BASTION_STEP)
}
export function sfxLedger() {
  playBlip(880, 0.08, 'square', 0.2)
  setTimeout(() => playBlip(1175, 0.1, 'square', 0.15), 80)
}
export function sfxRoot() {
  playBlip(120, 0.25, 'sawtooth', 0.25)
}
```

In `Level1_1.jsx`, replace `startLevelMusic = startTideMusic` with `startBastionMusic` and call `sfxLedger` on ledger stamp / `sfxRoot` when hedge roots.

## `src/characters/Spawn.jsx` — Hedge spawn model (append)

```jsx
export function HedgeSpawn({ e }) {
  const g = useRef()
  useFrame(({ clock }) => {
    if (!g.current || !e.alive) return
    const s = 0.35 + (e.rooting || 0) * 0.4
    g.current.scale.setScalar(s)
    g.current.rotation.y = clock.elapsedTime * 2 + e.seed
  })
  return (
    <group ref={g} position={[e.x, 0, e.z]}>
      <mesh position={[0, 0.25, 0]}>
        <icosahedronGeometry args={[0.28, 0]} />
        <meshStandardMaterial color="#1a2830" emissive="#8060d0" emissiveIntensity={1.2} flatShading toneMapped={false} />
      </mesh>
      <mesh position={[0, 0.05, 0]} rotation={[-Math.PI / 2, 0, 0]}>
        <circleGeometry args={[0.5, 12]} />
        <meshStandardMaterial color="#243830" transparent opacity={0.5} />
      </mesh>
    </group>
  )
}
```
